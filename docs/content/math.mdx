### isPrime


```ts tab='Source'
export function isPrime(value: number): boolean {
	if (value <= 1) return false;
	for (let i = 2; i <= Math.sqrt(value); i++) {
		if (value % i === 0) return false;
	}
	return true;
}
```

Checks whether a number is a prime number.

```ts title='Example'
isPrime(2); // true
isPrime(3); // true
isPrime(563); // true
isPrime(26881); // true

isPrime(1); // false
isPrime(4); // false
isPrime(564); // false
```

### isInRange

```ts tab='Shape'
function isInRange(value: number, range: RORange, inclusive: InclusiveMinMax = {}): boolean;

type RORange = readonly [start: number, end: number];
type InclusiveMinMax = {
	inclusiveMin?: boolean;
	inclusiveMax?: boolean;
};
```

```ts tab='Source'
export function isInRange(value: number, range: RORange, inclusive: InclusiveMinMax = {}): boolean {
	const [min, max] = range;
	const isGTMin = inclusive.inclusiveMin !== false ? value >= min : value > min;
	const isLTMax = inclusive.inclusiveMax !== false ? value <= max : value < max;
	return isGTMin && isLTMax;
}
```

Checks whether a number is within a given range.

<Callout type="info">Both `min` and `max` are inclusive by default.</Callout>

```ts title='Example'
const range = [1, 10] as const;

isInRange(0, range); // false

isInRange(1, range); // true
isInRange(5, range); // true
isInRange(10, range); // true

isInRange(11, range); // false
```

### doRangesOverlap

```ts tab='Shape'
function doRangesOverlap(ranges: RORange[], touchingCounts: boolean = false): boolean;

type RORange = readonly [start: number, end: number];
```

```ts tab='Source'
export function doRangesOverlap(ranges: RORange[], touchingCounts: boolean = false): boolean {
	if (ranges.some(([start, end]) => start > end)) throw new Error("Invalid range: start > end");

	const sorted = ranges.toSorted((a, b) => a[0] - b[0]);

	for (let index = 1; index < sorted.length; index++) {
		const range = sorted[index]!;
		const prevRange = sorted[index - 1]!;
		if (touchingCounts ? range[0] <= prevRange[1] : range[0] < prevRange[1]) return true;
	}

	return false;
}
```

Checks whether some ranges overlap.

<Callout type="info">
	By default, ranges sharing the same boundary are not considered overlapping. Set `touchingCounts`
	to `true` to consider them as overlapping.
</Callout>

<Callout type="warn">
	Throws a
	[RangeError](https://developer.mozilla.org/en-US/docs/Web/JavaScript/Reference/Global_Objects/RangeError)
	If any range has a start greater than its end.
</Callout>

```ts title='Example'
doRangesOverlap([
	[0, 3],
	[3, 6],
]); // false
doRangesOverlap([
	[0, 3],
	[2, 6],
]); // true

doRangesOverlap(
	[
		[0, 3],
		[3, 6],
	],
	true,
); // true

doRangesOverlap([
	[0, 3],
	[6, 3],
]); // throws // 6 > 3
```

### getRandomFloat

```ts tab='Shape'
function getRandomFloat(min: number, max: number): number;
```

```ts tab='Source'
export function getRandomFloat(min: number, max: number): number {
	if (min > max) [min, max] = [max, min];
	else if (min === max) return min;
	return Math.random() * (max - min + 1) + min;
}
```

Get a random float between the given range.

```ts title='Example'
getRandomFloat(1, 10);
3.1458241132191636;
getRandomFloat(1, 10);
2.076135543542401;
getRandomFloat(1, 10);
5.6216396763729986;
```
