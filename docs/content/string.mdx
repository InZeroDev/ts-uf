## format

### formatString

```ts tab='Shape'
function formatString<P extends string>(pattern: P, context: ExtractContext<P>): string;
```

```ts tab='Source'

export function formatString<P extends string>(pattern: P, context: ExtractContext<P>): string {
	return pattern.replace(/(\\*){(\w+)(?:=([^}]+))?}/g, (match, escapes, name, defaultValue) => {
		const newEscapesCount = Math.floor(escapes.length / 2);
		if (isOdd(escapes.length)) return match.slice(newEscapesCount + 1);
		return "\\".repeat(newEscapesCount) + (context[name as never] ?? defaultValue ?? match);
	});
}
```

Formats a string using a given context.

```ts title='Example'
formatString("Hello, {name}! Your age is {age=unknown}.", { name: "John" }); // "Hello, John! Your age is unknown."
```

## replace

### replaceSubstring

```ts tab='Shape'
function replaceSubstring(string: string, ...updates: ReplaceUpdate[]): string;

type ReplaceUpdate = [start: number, end: number, replacement: string];
```

```ts tab='Source'

export function replaceSubstring(string: string, ...updates: ReplaceUpdate[]): string {
	if (doRangesOverlap(updates.map(([start, end]) => [start, end])))
		throw new Error("Overlapping ranges");
	return updates.reduce(
		(acc, [start, end, replacement]) => {
			if (start < 0) throw new RangeError("Invalid range: start < 0");

			start += acc.delta;
			end += acc.delta;

			acc.string = acc.string.slice(0, start) + replacement + acc.string.slice(end);

			const rangeLength = end - start;
			acc.delta += replacement.length - rangeLength;

			return acc;
		},
		{ string, delta: 0 },
	).string;
}
```

Replaces a substring using its coordinates.

<Callout type='info'>
Automatically adjusts the ranges of subsequent updates to account for changes in string length
caused by earlier replacements, ensuring each update targets the intended substring.
</Callout>

<Callout type='warn'>
Throws a [RangeError](https://developer.mozilla.org/en-US/docs/Web/JavaScript/Reference/Global_Objects/RangeError) if:
- `start > end` or `start < 0`.
- some ranges overlap.
</Callout>

```ts title='Example'
replaceString("Hello World!", [6, 11, "Universe"]); // "Hello Universe!"
replaceString("012345678", [0, 2, "x"], [3, 5, "y"], [6, 8, "z"]); // "x2y5z8"
```
