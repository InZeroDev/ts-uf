## validation

### isBoolean

```ts tab='Shape'
function isBoolean<T>(value: CheckT<T>): value is SmartExtract<T, boolean>;
```

```ts tab='Source'

export function isBoolean<T>(value: CheckT<T>): value is SmartExtract<T, boolean> {
	return typeof value === "boolean";
}
```

Checks whether a value is a boolean.

```ts title='Example'
function foo(value: boolean | { a: boolean, b: boolean }) {
  if (isBoolean(value)) {
    value; // boolean
  }
}
```

### isNotBoolean

```ts tab='Shape'
function isNotBoolean<T>(value: CheckT<T>): value is Exclude<T, boolean>;
```

```ts tab='Source'

export function isNotBoolean<T>(value: CheckT<T>): value is Exclude<T, boolean> {
	return !isBoolean(value);
}
```

Checks whether a value is not a boolean.

```ts title='Example'
function foo(value: boolean | { a: boolean, b: boolean }) {
  if (isNotBoolean(value)) {
    value; // { a: boolean, b: boolean }
  }
}
```

### not

```ts tab='Shape'
function not(value: boolean): boolean;
```

```ts tab='Source'

export function not(value: boolean): boolean {
	return !value;
}
```

Negates a boolean.
