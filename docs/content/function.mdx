## validation

### isFunction

```ts tab='Shape'
function isFunction<T>(value: CheckT<T>): value is SmartExtract<T, Func>;
```

```ts tab='Source'
export function isFunction<T>(value: CheckT<T>): value is SmartExtract<T, Func> {
	return typeof value === "function";
}
```

Checks whether a value is a function.

```ts title='Example'
function foo(valueOrFn: string | (() => string)) {
	const value = isFunction(valueOrFn) ? valueOrFn() : valueOrFn;
}
```

```ts tab='Source'
export function isNotFunction<T>(value: CheckT<T>): value is Exclude<T, Func> {
```

```ts tab='Shape'
function repeat(func: RepeatCb, data: RepeatData): void;

type RepeatCb = Func<RepeatCbParams, RepeatCbRT>;
type RepeatCbParams = [index: number];
```

```ts tab='Source'
export function repeat(func: RepeatCb, data: RepeatData): void;
export function repeat(func: RepeatCb, times: number): void;
export function repeat(func: RepeatCb, arg2: number | RepeatData): void {
	const data = argToData(arg2);

	let index = data.start;
	for (; index < data.end;) {
		const result = func(index);
		if (result === false) break;
		if (isNumber(result)) index = result;
		else index += data.step;
	}
}
```

Repeats a function a given number of times.

<Callout type='info'>

- Return `false` to break the loop.
- Return a `number` to jump to that index.

</Callout>

```ts title='Example'
repeat(console.log, { start: 10, end: 13 }); // logs "10", "11", "12"
repeat(console.log, { start: 10, count: 3 }); // logs "10", "11", "12"

repeat((index) => {
	if (index === 3) return 5; // jump to index 5
	if (index === 8) return false; // break the loop
}, 10);
```

### repeatWhile

```ts tab='Shape'
function repeatWhile(func: RepeatCb, condition: RepeatWhileCondition): void;

type RepeatCb = Func<RepeatCbParams, RepeatCbRT>;
type RepeatCbRT = boolean | number | void;

type RepeatWhileCondition = Func<RepeatCbParams, boolean>;
type RepeatCbParams = [index: number];
```

```ts tab='Source'
export function repeatWhile(func: RepeatCb, condition: RepeatWhileCondition): void {
	let index = 0;
	while (condition(index)) {
		const result = func(index);
		if (result === false) break;
		if (isNumber(result)) index = result;
		else index++;
	}
}
```

Repeats a function while a given condition is true.

<Callout type='info'>

- Return `false` to break the loop.
- Return a `number` to jump to that index.

</Callout>

```ts title='Example'
repeatWhile(console.log, (index) => index < 3); // logs "0", "1", "2"
```

## utils

### iife

```ts tab='Shape'
function iife<const A extends ROList, const R>(func: Func<A, R>, ...args: A): R;
```

```ts tab='Source'
export function iife<const A extends ROList, const R>(func: Func<A, R>, ...args: A): R {
	return func(...args);
}
```

Immediately invokes a function and returns its result.
Useful for creating a local scope to organize your code.

```ts title='Example'
function cliAction() {
	const folders = iife(() => {
		const source = "...";
		const options = { ... };
		const folders = glob.sync(source, options);
		return folders.filter(...);
	});
	const files = iife(() => {
		const source = "...";
		const options = { ... };
		const files = glob.sync(source, options);
		return files.filter(...);
	});
}
```


```ts tab='Source'
export async function sleep(ms: number): Promise<void> {
	return new Promise((resolve) => setTimeout(resolve, ms));
}
```

Sleeps for the specified number of milliseconds.

```ts tab='Shape'
async function withDelay(func: Func, ms: number): Promise<void>;
```

```ts tab='Source'
export async function withDelay(func: Func, ms: number): Promise<void> {
```
