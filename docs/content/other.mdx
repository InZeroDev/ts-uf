## validation

### isOneOf

```ts tab='Shape'
function isOneOf<const T, const L extends ROList>(
	value: T,
	list: L,
): value is SmartExtract<T, L[number]>;
```

```ts tab='Source'

export function isOneOf<const T, const L extends ROList>(
	value: T,
	list: L,
): value is SmartExtract<T, L[number]> {
	return list.includes(value as Valid);
}
```

checks whether a value is one of the expected values.

```ts title='Example'
const colors = ["red", "green", "blue"] as const;

function foo(color: string) {
  if (isOneOf(color, colors)) {
    color; // "red" | "green" | "blue"
  }
}
```

### isNotOneOf

```ts tab='Shape'
function isNotOneOf<const T, const L extends ROList>(
	value: T,
	list: L,
): value is Exclude<T, L[number]>;
```

```ts tab='Source'

export function isNotOneOf<const T, const L extends ROList>(
	value: T,
	list: L,
): value is Exclude<T, L[number]> {
	return !isOneOf(value, list);
}
```

Checks whether a value is not one of the expected values.

```ts title='Example'
type Action = "start" | "stop" | "pause";

function foo(action: string) {
  if (isNotOneOf(action, ["stop", "pause"])) {
    startSomething();
    action; // "start"
  }
}
```

### isEmpty

```ts tab='Shape'
function isEmpty<T>(value: _CheckT<T, string | ROList | Dict>): value is IsEmptyRT<T>;
```

```ts tab='Source'

export function isEmpty<T>(value: _CheckT<T, string | ROList | Dict>): value is IsEmptyRT<T> {
	const v: unknown = value;
	if (isString(v)) return isEmptyString(v);
	if (isArray(v)) return isEmptyArray(v);
	if (isObject(v)) return isEmptyObject(v);
	throw new Error("Invalid type");
}
```

A wrapper for [isEmptyString](#isemptystring), [isEmptyArray](#isemptyarray) and [isEmptyObject](#isemptyobject).

## utils

### mapDefined

```ts tab='Shape'
function mapDefined<T, const R>(
	value: T,
	fn: Func<[value: Exclude<T, undefined | null>], R>,
): R | Extract<T, undefined | null>;
```

```ts tab='Source'

export function mapDefined<T, const R>(
	value: T,
	fn: Func<[value: Exclude<T, undefined | null>], R>,
): R | Extract<T, undefined | null> {
	return isOneOf(value, [undefined, null]) ? (value as Valid) : fn(value as Valid);
}
```

Maps a value to another value if it is not `undefined` or `null`.

```ts title='Example'
// Instead of:
function getDescription(func: FunctionDeclaration) {
  const description = func.getJsDocs().at(-1)?.getDescription();
  if (description === undefined) return; // Notice that we used `description === undefined` and not `!description`.
  return convertJsDocLinks(description);
}
// You can do:
function getDescription(func: FunctionDeclaration) {
  return mapDefined(func.getJsDocs().at(-1)?.getDescription(), convertJSDocLinks);
}
```
