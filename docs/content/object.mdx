### setDeep

```ts tab='Shape'
function setDeep<T extends Dict, K extends Suggest<KeysDeep<T>>, V>(
	obj: T,
	key: K,
	value: V,
): void;
```

```ts tab='Source'
export function setDeep<T extends Dict, K extends Suggest<KeysDeep<T>>, V>(
	obj: T,
	key: K,
	value: V,
): void {
	const keys = key.split(".");
	const lastKey = keys.pop();
	if (lastKey === undefined) throw new Error("Invalid key");

	const lastObj = keys.reduce((acc, currKey, ind) => {
		const currValue = (acc as Valid)?.[currKey];
		if (isNotObject(currValue)) {
			const invalidKey = keys.slice(0, ind + 1).join(".");
			throw new Error(`${invalidKey} is not an object.`);
		}
		return currValue;
	}, obj) as Dict;
	lastObj[lastKey] = value;
}
```

Sets a deeply nested value in an object using a dot-separated key.

```ts title='Example'
const data = { a: 1, b: { c: 4 } };

setDeep(cloneDeep(data), "b.c", 47); // { a: 1, b: { c: 47 } }
setDeep(cloneDeep(data), "b.p", "new"); // { a: 1, b: { c: 4, p: "new" } }

setDeep(cloneDeep(data), "a.y", "new"); // Error: "a" is not an object.
```

## merge

### mergeIntersection

```ts tab='Shape'
function mergeIntersection<A extends Dict, B extends Dict>(a: A, b: B): MergeInt<A, B>;
```

```ts tab='Source'
export function mergeIntersection<A extends Dict, B extends Dict>(a: A, b: B): MergeInt<A, B> {
	const [aKeys, bKeys] = [getKeys(a), getKeys(b)];
	const keysToDelete = bKeys.filter((bKey) => isNotOneOf(bKey, aKeys));

	const result = updateValues(deleteKeys(b, keysToDelete), (value, key) =>
		isUndefined(value) ? a[key] : value,
	);

	return result as Valid;
}
```

Merge two objects, keeping all keys that exist in both.
Values are taken from `B` when defined, otherwise falling back to `A`.

```ts title='Example'
mergeIntersection({ a: 1, b: 2 }, { b: 22, c: 4 }); // { b: 22 }
mergeIntersection({ a: 1, b: 2 }, { b: undefined, c: 4 }); // { b: 2 }
mergeIntersection({ a: 1, b: undefined }, { b: undefined, c: 4 }); // { b: undefined }
```

```ts tab='Source'
export function mergeDefaults<S extends Dict>(
	source: S,
	defaults: Prettify<Required<PickOptional<S> & PickUndefinable<S>>>,

	return {
		...updateValues(source, (value, key) => (isUndefined(value) ? (defaults as Dict)[key] : value)),
		...mergeDifference(defaults, source),
	} as Valid;
}
```

Merges default values into a source object, using them for missing or undefined properties.

```ts title='Example'
type Config = { a: number; b?: number; c: number | undefined; d?: number };

const config: Config = {
	a: 1,
	b: 2,
	c: undefined,
```

### mergeDefaultsDeep

```ts tab='Shape'
function mergeDefaultsDeep<S extends Dict>(
	source: S,
	defaults: RequiredDeep<PickOptionalDeep<S> & PickUndefinableDeep<S>>,
): MergeDefaultsDeepRT<S>;
```

```ts tab='Source'
export function mergeDefaultsDeep<S extends Dict>(
	source: S,
	defaults: RequiredDeep<PickOptionalDeep<S> & PickUndefinableDeep<S>>,
): MergeDefaultsDeepRT<S> {
	return {
		...updateValues(source, (value, key) => {
			const df = defaults as Dict;
			if (isUndefined(value)) return df[key];
			if (isObject(value) && isObject(df[key]))
				return mergeDefaultsDeep(value as Dict, df[key] as Valid);
			return source[key];
		}),
		...mergeDifference(defaults, source),
	} as Valid;
}
```

Recursively merges default values into a source object, using them for missing or undefined properties.

```ts title='Example'
type Config = { a?: number; b: { c: number; d: number | undefined } };
const config: Config = { b: { c: 4, d: undefined } };

const fullConfig = mergeDefaultsDeep(config, { a: 1, b: { d: 99 } }); // { a: 1, b: { c: 4, d: 99 } }
```
